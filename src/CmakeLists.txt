        CUDA::cudart

target_compile_definitions(Simulation PRIVATE SIM_WITH_IMGUI=1)

target_compile_features(Simulation PRIVATE cxx_std_20)

set_target_properties(Simulation PROPERTIES
    CUDA_SEPARABLE_COMPILATION ON
    CUDA_STANDARD 17
    CUDA_STANDARD_REQUIRED ON
)

if(MSVC)
    target_compile_options(Simulation PRIVATE
        # C++ файлы — флаги напрямую
        $<$<COMPILE_LANGUAGE:CXX>:/openmp:llvm>
        $<$<COMPILE_LANGUAGE:CXX>:/arch:AVX2>
        $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:/O2>
        $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:/fp:fast>

        # CUDA файлы — флаги через -Xcompiler для host-части
        $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/openmp:llvm>
        $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/arch:AVX2>
        $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CONFIG:Release>>:-Xcompiler=/O2>
        $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CONFIG:Release>>:-Xcompiler=/fp:fast>
    )
else()
    target_compile_options(Simulation PRIVATE
        $<$<COMPILE_LANGUAGE:CXX>:-march=native>
        $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-O3>
        $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-ffast-math>

        $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-march=native>
        $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CONFIG:Release>>:-Xcompiler=-O3>
        $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CONFIG:Release>>:-Xcompiler=-ffast-math>
    )
endif()

target_include_directories(Simulation
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)

add_custom_command(TARGET Simulation POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_SOURCE_DIR}/shaders"
        "$<TARGET_FILE_DIR:Simulation>/shaders"
)

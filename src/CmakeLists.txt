# Исполняемый файл
add_executable(Simulation
    main.cpp

    app/app.cpp
    app/app.h

    common/config.h

    data/particleData.h
    data/particleData.cpp

    gui/userInterface.cpp
    gui/userInterface.h

    render/renderer.cpp
    render/renderer.h

    input/inputManager.cpp
    input/inputManager.h

    sim/engine.cpp
    sim/engine.h
    sim/structs.h

    sim/collisions/pairsGrid.cpp
    sim/collisions/pairsGrid.h
    sim/collisions/pairsNaive.cpp
    sim/collisions/pairsNaive.h

    sim/constraints/boxBounds.cpp
    sim/constraints/boxBounds.h
    sim/constraints/circleCollision.cpp
    sim/constraints/circleCollision.h
)

# Линковка библиотек (имена — как их даёт vcpkg find_package в корневом CMake)
target_link_libraries(Simulation
    PRIVATE
        glfw               # glfw3::glfw может называться просто glfw 
        glad::glad         # загрузчик OpenGL 
        glm::glm           # математика для рендера 
        Eigen3::Eigen      # математика для физики 
        OpenGL::GL         # сам OpenGL 
)


target_link_libraries(Simulation PRIVATE imgui::imgui)
target_compile_definitions(Simulation PRIVATE SIM_WITH_IMGUI=1)


target_compile_features(Simulation PRIVATE cxx_std_20)

# Общие include-пути, чтобы можно было писать #include "app/app.h" и т.п.
target_include_directories(Simulation
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)

# Копирование шейдеров к .exe
add_custom_command(TARGET Simulation POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_SOURCE_DIR}/shaders"
    "$<TARGET_FILE_DIR:Simulation>/shaders"
)
